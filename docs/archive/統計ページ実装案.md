## 💾 必要なAPIエンドポイント

### 新規作成が必要なAPI（4つ）

#### 1. GET /api/stats/summary
基本統計情報を取得

**レスポンス例**:
```json
{
  "total_spots": 21,
  "total_reviews": 38,
  "total_users": 4,
  "total_events": 20,
  "avg_rating_overall": 4.2
}
```

**SQL例**:
```sql
SELECT
  (SELECT COUNT(*) FROM tourist_spots) as total_spots,
  (SELECT COUNT(*) FROM reviews) as total_reviews,
  (SELECT COUNT(*) FROM users) as total_users,
  (SELECT COUNT(*) FROM events) as total_events,
  (SELECT AVG(avg_rating) FROM tourist_spots) as avg_rating_overall
```

#### 2. GET /api/stats/spots-by-area
地域別観光地数を取得

**レスポンス例**:
```json
[
  {"area": "maebashi", "area_name": "前橋・赤城", "count": 6},
  {"area": "takasaki", "area_name": "高崎・富岡", "count": 4},
  {"area": "kusatsu", "area_name": "草津・四万", "count": 3},
  {"area": "minakami", "area_name": "水上・尾瀬", "count": 3},
  {"area": "ikaho", "area_name": "伊香保・榛名", "count": 2},
  {"area": "other", "area_name": "その他", "count": 3}
]
```

**実装ポイント**:
- addressカラムから地域を判定（spots.jsと同じロジック）
- または、地域カラムを追加

#### 3. GET /api/stats/events-by-month
月別イベント数を取得

**レスポンス例**:
```json
[
  {"month": 1, "month_name": "1月", "count": 2},
  {"month": 2, "month_name": "2月", "count": 1},
  {"month": 3, "month_name": "3月", "count": 1},
  {"month": 4, "month_name": "4月", "count": 2},
  {"month": 5, "month_name": "5月", "count": 2},
  {"month": 6, "month_name": "6月", "count": 1},
  {"month": 7, "month_name": "7月", "count": 2},
  {"month": 8, "month_name": "8月", "count": 4},
  {"month": 9, "month_name": "9月", "count": 1},
  {"month": 10, "month_name": "10月", "count": 2},
  {"month": 11, "month_name": "11月", "count": 1},
  {"month": 12, "month_name": "12月", "count": 1}
]
```

**SQL例**:
```sql
SELECT
  CAST(substr(event_date, 6, 2) AS INTEGER) as month,
  COUNT(*) as count
FROM events
GROUP BY month
ORDER BY month
```

#### 4. GET /api/stats/top-spots?limit=5
人気観光地ランキング取得

**クエリパラメータ**:
- `limit`: 取得件数（デフォルト: 5）

**レスポンス例**:
```json
[
  {
    "spot_id": 15,
    "spot_name": "こんにゃくパーク",
    "avg_rating": 4.5,
    "review_count": 4
  },
  {
    "spot_id": 1,
    "spot_name": "草津温泉",
    "avg_rating": 4.7,
    "review_count": 3
  }
]
```

**SQL例**:
```sql
SELECT spot_id, spot_name, avg_rating, review_count
FROM tourist_spots
WHERE review_count > 0
ORDER BY avg_rating DESC, review_count DESC, spot_id ASC
LIMIT ?
```

**代替案**: 既存の `GET /api/spots` を使ってフロントエンドでソートも可能

### JavaScript（stats.js）

```javascript
// ページ読み込み時に実行
async function loadStats() {
  try {
    // 並列で全データ取得
    const [summary, topSpots, areaData, monthData] = await Promise.all([
      apiClient.getStatsSummary(),
      apiClient.getTopSpots(5),
      apiClient.getStatsByArea(),
      apiClient.getStatsByMonth()
    ]);

    displaySummary(summary);
    displayTopSpots(topSpots);
    displaySpotsByArea(areaData);
    displayEventsByMonth(monthData);
  } catch (error) {
    console.error('統計データの読み込みエラー:', error);
    alert('統計データの読み込みに失敗しました');
  }
}

// 棒グラフ描画関数
function renderBarChart(containerId, data, labelKey, valueKey) {
  const container = document.getElementById(containerId);
  const maxValue = Math.max(...data.map(d => d[valueKey]));

  data.forEach(item => {
    const barItem = document.createElement('div');
    barItem.className = 'bar-chart-item';

    const width = (item[valueKey] / maxValue) * 100;

    barItem.innerHTML = `
      <span class="bar-chart-label">${item[labelKey]}</span>
      <div class="bar-chart-bar" style="width: ${width}%"></div>
      <span class="bar-chart-value">${item[valueKey]}</span>
    `;

    container.appendChild(barItem);
  });
}
```
